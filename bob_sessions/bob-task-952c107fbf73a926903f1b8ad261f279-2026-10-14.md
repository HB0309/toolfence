# ---------------------------------------------------------------------------
# Helpers
# ---------------------------------------------------------------------------

def _make_surface(*entries: SurfaceEntry) -> SurfaceFile:
    return SurfaceFile(version=1, tools=list(entries))


def _entry(
    name: str,
    *,
    action_class: str = "read",
    trust_label: str = "first-party",
    description: str = "A test tool.",
) -> SurfaceEntry:
    return SurfaceEntry(
        name=name,
        route=f"GET /{name}",
        action_class=action_class,  # type: ignore[arg-type]
        trust_label=trust_label,  # type: ignore[arg-type]
        description_sha256=description_sha256(description),
    )


def _write_surface(surface: SurfaceFile) -> Path:
    """Write *surface* to a temp file and return its path."""
    tmp = tempfile.NamedTemporaryFile(
        mode="w", suffix=".json", delete=False, encoding="utf-8"
    )
    tmp.write(json.dumps(surface.model_dump()))
    tmp.close()
    return Path(tmp.name)


# ---------------------------------------------------------------------------
# Stub registry for guard_wiring_check tests

class _StubTool:
    def __init__(self, name: str) -> None:
        self.name = name

    async def execute(self, **params: Any) -> Any:
        return "ok"


class _StubRegistry:
    def __init__(self, tools: dict[str, Any]) -> None:
        self._tools = tools

    @property
    def tool_names(self) -> list[str]:
        return list(self._tools.keys())

# CLI-5 test 1 — unknown tool in live list
# ===========================================================================

def test_surface_diff_unknown_tool() -> None:
    """Live list contains a tool not present in surface → exit 1, tool name in message."""
    known_name = "mcp_galaxium_get_flights_flights_get"
    unknown_name = "mcp_galaxium_new_mystery_endpoint_post"

    surface = _make_surface(_entry(known_name))
    surface_path = _write_surface(surface)

    live_tool_list = [
        {"name": known_name, "description": "A test tool.", "method": "GET"},
        {"name": unknown_name, "description": "A brand new tool.", "method": "POST"},
    ]

    exit_code, message = surface_diff_check(surface_path, live_tool_list)

    assert exit_code == 1, (
        f"Expected exit code 1 for unknown tool, got {exit_code}. Message: {message}"
    )
    assert unknown_name in message, (
        f"Expected unknown tool name '{unknown_name}' in message.\nGot: {message}"
    )

# CLI-5 test 2 — changed description hash
# ===========================================================================

def test_surface_diff_changed_hash() -> None:
    """Live tool has a different description → description drift → exit 1, tool name in message."""
    tool_name = "mcp_galaxium_get_flights_flights_get"
    original_description = "Original description for the tool."

    # Surface is committed with a hash of original_description.
    surface = _make_surface(
        SurfaceEntry(
            name=tool_name,
            route="GET /flights",
            action_class="read",  # type: ignore[arg-type]
            trust_label="first-party",  # type: ignore[arg-type]
            description_sha256=description_sha256(original_description),
        )
    )
    surface_path = _write_surface(surface)

    # Live list has the same tool but a different description.
    live_tool_list = [
        {
            "name": tool_name,
            "description": "CHANGED description — someone edited the tool.",
            "method": "GET",
        }
    ]

    exit_code, message = surface_diff_check(surface_path, live_tool_list)

    assert exit_code == 1, (
        f"Expected exit code 1 for hash drift, got {exit_code}. Message: {message}"
    )
    assert tool_name in message, (
        f"Expected tool name '{tool_name}' in message.\nGot: {message}"
    )

**There are three ways, in order of effort:**

### a) Push the branch to GitHub (the real thing)

[`act`](https://github.com/nektos/act) runs GitHub Actions locally via Docker. From the repo root:
```powershell
act push
```
This is the closest local simulation of the full Action, but requires Docker.

## 2. Should you fix the Node.js 20 deprecation warning?

### What the warning means

### What you actually need to change
Pin to the newer versions of those actions that already target Node 24:

| Current | Fix |
|---|---|
| `actions/checkout@v4` | `actions/checkout@v4.2.2` (or `@v4` with a newer minor — already Node 24 compatible in recent point releases) |
| `actions/setup-python@v5` | `actions/setup-python@v5.3.0` or later |

import argparse
import json
from http.server import BaseHTTPRequestHandler, HTTPServer
from pathlib import Path

# ---------------------------------------------------------------------------
# Paths
# ---------------------------------------------------------------------------

_HERE = Path(__file__).parent
_STATIC_DIR = _HERE / "static"


# ---------------------------------------------------------------------------
# Request handler

class _DashboardHandler(BaseHTTPRequestHandler):
    """HTTP request handler for the ToolFence dashboard."""

    # Injected by the server factory before the first request.
    results_dir: Path = Path("results")

    def log_message(self, format: str, *args: object) -> None:  # noqa: A002
        """Silence the default per-request log to keep stdout clean."""

    def do_GET(self) -> None:  # noqa: N802
        path = self.path.split("?")[0]  # strip query string

        if path in ("/", ""):

    # ------------------------------------------------------------------

        if not filepath.is_file():
            self._not_found()
            return
        data = filepath.read_bytes()
        self.send_response(200)
        self.send_header("Content-Type", content_type)
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def _serve_results(self) -> None:
        records = _load_records(self.results_dir)
        body = json.dumps(records, default=str).encode("utf-8")
        self._send_json(body)

        self.send_response(200)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

        self.send_response(404)
        self.end_headers()
        self.wfile.write(b"Not found")

def _mime(filename: str) -> str:
    if filename.endswith(".html"):
        return "text/html; charset=utf-8"
    if filename.endswith(".js"):
        return "application/javascript; charset=utf-8"
    if filename.endswith(".css"):

    return "application/octet-stream"

        except Exception:  # noqa: BLE001

function esc(v) {
  if (v === null || v === undefined) return "—";
  return String(v)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;");
}

function guardBadge(on) {
  return on
    ? '<span class="badge badge-on">on</span>'
    : '<span class="badge badge-off">off</span>';
}

function verdictBadge(v) {
  if (!v) return "—";
  const cls = {
    leaked: "badge-leaked",
    held: "badge-held",
    refused: "badge-refused",
    inconclusive: "badge-inconclusive",
  }[v] || "";
  return `<span class="badge ${cls}">${esc(v)}</span>`;
}

function pct(n, total) {
  if (!total) return "—";
  return (n / total * 100).toFixed(1) + "%";
}

        <th class="num">Skipped</th>

        <th class="num">Leaked%</th>

      </tr></thead>

  const rows = conditions.map(c => {

// ---------------------------------------------------------------------------
// Trend table — one row per distinct run_id in /api/results
// ---------------------------------------------------------------------------

function buildTrendTable(records) {
  // Group by run_id, count verdicts per run
  const runs = {};
  for (const r of records) {
    const rid = r.run_id || "unknown";
    if (!runs[rid]) runs[rid] = { leaked: 0, held: 0, refused: 0, inconclusive: 0, total: 0 };
    const run = runs[rid];
    run.total++;
    if (r.kind === "attack") {
      if (r.verdict === "leaked")       run.leaked++;
      else if (r.verdict === "held")    run.held++;
      else if (r.verdict === "refused") run.refused++;
      else if (r.verdict === "inconclusive") run.inconclusive++;
    }
  }

  const runIds = Object.keys(runs);
  if (runIds.length === 0) {
    return "<p style='color:#57606a;font-style:italic'>No run data available.</p>";
  }

  let html = `
    <table class="trend-table">
      <thead><tr>
        <th>Run ID</th>
        <th class="num">Total</th>
        <th class="num">Leaked</th>
        <th class="num">Held</th>
        <th class="num">Refused</th>
        <th class="num">Inconclusive</th>
      </tr></thead>
      <tbody>`;

  for (const rid of runIds) {
    const run = runs[rid];
    html += `<tr>
      <td class="run-id">${esc(rid)}</td>
      <td class="num">${run.total}</td>
      <td class="num">${run.leaked}</td>
      <td class="num">${run.held}</td>
      <td class="num">${run.refused}</td>
      <td class="num">${run.inconclusive}</td>
    </tr>`;
  }

  html += "</tbody></table>";
  if (runIds.length < 2) {
    html += `<p style="font-size:0.8rem;color:#57606a;margin-top:4px">
      (Trend view shows multiple runs once more than one run_id is present in results/)
    </p>`;
  }
  return html;
}

let _replayRecords = [];
let _replayIndex = 0;
let _replayTimer = null;
const REPLAY_INTERVAL_MS = 800;

function initReplay(records) {
  _replayRecords = records;
  _replayIndex = 0;

  const btn = document.getElementById("replay-btn");
  const counter = document.getElementById("replay-counter");
  const progress = document.getElementById("replay-progress");

  if (!records || records.length === 0) {
    btn.disabled = true;
    counter.textContent = "No records to replay.";
    return;
  }

  btn.disabled = false;
  counter.textContent = `0 / ${records.length} records`;
  progress.innerHTML = "";

  btn.addEventListener("click", function () {
    if (_replayTimer) {
      clearInterval(_replayTimer);
      _replayTimer = null;
      btn.textContent = "▶ Start Replay";
      return;
    }
    // Reset
    _replayIndex = 0;
    progress.innerHTML = "";
    counter.textContent = `0 / ${records.length} records`;
    btn.textContent = "⏸ Pause";

    _replayTimer = setInterval(function () {
      if (_replayIndex >= _replayRecords.length) {
        clearInterval(_replayTimer);
        _replayTimer = null;
        btn.textContent = "▶ Start Replay";
        return;
      }
      const rec = _replayRecords[_replayIndex];
      _replayIndex++;

      counter.textContent = `${_replayIndex} / ${records.length} records`;

      const card = document.createElement("div");
      card.className = "replay-card";
      card.setAttribute("data-trial-index", _replayIndex);
      card.innerHTML = `
        <span class="label">trial_id</span>
        <span class="value">${esc(rec.trial_id)}</span>
        &nbsp;|&nbsp;
        <span class="label">kind</span>
        <span class="value">${esc(rec.kind)}</span>
        &nbsp;|&nbsp;
        <span class="label">defense</span>
        <span class="value">${esc(rec.defense)}</span>
        &nbsp;|&nbsp;
        <span class="label">guard</span>
        ${guardBadge(rec.guard_on)}
        &nbsp;|&nbsp;
        <span class="label">verdict</span>
        ${rec.kind === "attack" ? verdictBadge(rec.verdict) : "<span style='color:#57606a'>—</span>"}
        ${rec.channel ? `&nbsp;|&nbsp;<span class="label">channel</span> <span class="value">${esc(rec.channel)}</span>` : ""}
      `;
      progress.prepend(card);
    }, REPLAY_INTERVAL_MS);
  });
}

async function bootstrap() {
  // Fetch summary for grids
  let summary = null;
  try {
    const resp = await fetch("/api/summary");

    document.getElementById("normal-loading").className = "error";
    document.getElementById("normal-loading").textContent = "Failed to load summary.";
  }

  if (summary) {
    document.getElementById("security-loading").style.display = "none";
    document.getElementById("security-grid").innerHTML =
      buildSecurityGrid(summary.conditions);

    document.getElementById("normal-loading").style.display = "none";
    document.getElementById("normal-grid").innerHTML =
      buildNormalGrid(summary.normal_conditions, summary.normal_tasks);
  }

  // Fetch raw records for trend + replay
  let records = [];
  try {
    const resp = await fetch("/api/results");
    records = await resp.json();
  } catch (e) {
    document.getElementById("trend-note").className = "error";
    document.getElementById("trend-note").textContent = "Failed to load results: " + e;
  }

  if (records) {
    document.getElementById("trend-note").style.display = "none";
    document.getElementById("trend-grid").innerHTML = buildTrendTable(records);
    initReplay(records);
  }
}

def test_dashboard_no_outbound_requests(dashboard_url: str) -> None:
    """No request leaves localhost during navigation and replay (CLI-6)."""
    outbound_urls: list[str] = []

    with sync_playwright() as p:
        browser = p.chromium.launch()
        page = browser.new_page()

        # Intercept every network request; flag anything not going to localhost/127.0.0.1
        def _on_request(request) -> None:
            url = request.url
            if not (
                url.startswith("http://localhost")
                or url.startswith("http://127.0.0.1")
                or url.startswith("data:")
            ):
                outbound_urls.append(url)

        page.on("request", _on_request)

        page.goto(dashboard_url)
        page.wait_for_load_state("networkidle")

        # Also click replay to exercise the JS path
        btn = page.query_selector("#replay-btn")
        if btn and not btn.is_disabled():
            btn.click()
            time.sleep(2)  # let a couple of steps fire

        browser.close()

    assert not outbound_urls, (
        "Dashboard made outbound requests (must only call localhost):\n"
        + "\n".join(outbound_urls)
    )
