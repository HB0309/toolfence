def test_no_existing_bob_file_modified():
    """Assert that no .bob/rules-* file has been changed since the commit that added it.

    Strategy: for each .bob/rules-* file, find the commit that first added it via
    ``git log --diff-filter=A``, then run ``git diff <add-commit> -- <file>``
    (without HEAD) to compare that commit against the working tree. This catches
    both committed and uncommitted modifications.
    """
    bob_rules_files = sorted((_REPO_ROOT / ".bob").glob("rules-*/**/*"))
    # Keep only files (not directories)
    bob_rules_files = [p for p in bob_rules_files if p.is_file()]
    assert bob_rules_files, "Expected at least one .bob/rules-* file to check"

    modified: list[str] = []

    for fpath in bob_rules_files:
        rel = fpath.relative_to(_REPO_ROOT).as_posix()

        # Find the commit that first added this file.
        log_result = subprocess.run(
            ["git", "log", "--diff-filter=A", "--name-only", "--format=%H", "--", rel],
            capture_output=True, text=True, cwd=str(_REPO_ROOT),
        )
        commits = [line.strip() for line in log_result.stdout.splitlines() if len(line.strip()) == 40]
        if not commits:
            # File not yet committed — treat as modified.
            modified.append(f"{rel}: not yet committed")
            continue

        add_commit = commits[-1]  # oldest (first) commit that added the file

        # Diff from that commit to the working tree (no HEAD argument).
        diff_result = subprocess.run(
            ["git", "diff", add_commit, "--", rel],
            capture_output=True, text=True, cwd=str(_REPO_ROOT),
        )
        if diff_result.stdout.strip():
            modified.append(f"{rel}: differs from add-commit {add_commit[:8]}")

    assert not modified, (
        "The following .bob/rules-* files have been modified since they were added:\n"
        + "\n".join(f"  {m}" for m in modified)
    )
