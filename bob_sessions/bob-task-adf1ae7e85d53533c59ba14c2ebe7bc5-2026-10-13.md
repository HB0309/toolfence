> 🔧 **write_file** — """Guard-wired async launcher for direct / eval nanobot runs (B-3).

Usage
-----
    python scripts/launch_nanobot.py \\
        --config config/nanobot.spike.json \\
        --prompt "Find the cheapest flight from Earth to Mars." \\
        --session-key "eval:alice:1"

The launcher:
1. Loads nanobot config and creates a fresh temp workspace.
2. Instantiates the full Guard stack (ProvenanceStore, SessionTaint,
   GuardDecision, ConfirmGrant).
3. Connects MCP servers via MCPProvider.
4. Calls install_guard to wrap every registry tool with GuardedProxy.
5. Calls provenance.record_user_message(prompt) so user-supplied values
   are resolved as "user" provenance (not untrusted) on the first turn.
6. Calls verify_guard_invariant to assert all tools are wrapped.
7. Runs agent_loop.process_direct and returns the response.
8. Tears down cleanly in a finally block.

This script is the basis for NanobotAdapter (B-4).  It intentionally has no
MessageBus subscriber — process_direct builds InboundMessage internally and
never goes through the bus inbound queue.
"""
from __future__ import annotations

import argparse
import asyncio
import shutil
import sys
import tempfile
from typing import Any

from toolfence.guard.confirm import ConfirmGrant
from toolfence.guard.decision import GuardDecision
from toolfence.guard.provenance import ProvenanceStore
from toolfence.guard.taint import SessionTaint
from toolfence.integration.installer import install_guard, verify_guard_invariant


def _noop_on_held(event: Any) -> None:
    """Placeholder held-call handler — prints a warning; B-4b wires the real one."""
    print(
        f"[warn] tool '{event.tool_name}' held (reason={event.reason}); "
        "no channel adapter installed — call will not be retried.",
        file=sys.stderr,
    )


async def _run(
    config_path: str,
    prompt: str,
    session_key: str,
    *,
    guard_on: bool = True,
    datamark_on: bool = False,
) -> dict[str, Any]:
    """Run one agent turn with Guard wiring and return a result dict."""
    from nanobot.agent.loop import AgentLoop
    from nanobot.agent.tools.mcp import MCPProvider
    from nanobot.agent.tools.registry import ToolRegistry
    from nanobot.bus.queue import MessageBus
    from nanobot.cli.runtime_config import _load_runtime_config
    from nanobot.providers.factory import make_provider

    from toolfence.contracts.surface import SurfaceFile

    workspace = tempfile.mkdtemp(prefix="toolfence-launch-")
    try:
        runtime_config = _load_runtime_config(config_path, workspace)

        # Guard stack — one instance per session.
        provenance = ProvenanceStore()
        taint = SessionTaint()
        decision = GuardDecision()
        confirm_grant = ConfirmGrant()

        # Surface map — load from the committed baseline.
        # B-5 will populate this with all tools including builtins.
        surface_map = SurfaceFile(version=1, tools=[])

        bus = MessageBus()
        tools = ToolRegistry()
        mcp_provider = MCPProvider.from_config(runtime_config, tools)
        provider = make_provider(runtime_config)

        agent_loop = AgentLoop.from_config(
            runtime_config,
            bus,
            provider=provider,
            tool_registry=tools,
        )

        try:
            await mcp_provider.connect()

            # Install Guard over every tool now in the registry.
            install_guard(
                tools,
                surface_map,
                provenance,
                taint,
                decision,
                _noop_on_held,
                confirm_grant,
                guard_on=guard_on,
                datamark_on=datamark_on,
            )

            # Record the user's prompt so values the user typed are resolved as
            # "user" provenance (not untrusted) inside GuardedProxy.
            provenance.record_user_message(prompt)

            # Per-turn invariant check — every tool must be wrapped.
            verify_guard_invariant(tools)

            response = await agent_loop.process_direct(
                prompt,
                session_key=session_key,
            )
        finally:
            try:
                await agent_loop.aclose()
            finally:
                await mcp_provider.aclose()
    finally:
        shutil.rmtree(workspace, ignore_errors=True)

    return {
        "response": response.content if response else None,
        "session_key": session_key,
    }


def main() -> None:
    parser = argparse.ArgumentParser(description="Guard-wired nanobot launcher (B-3).")
    parser.add_argument("--config", default="config/nanobot.spike.json")
    parser.add_argument("--prompt", required=True)
    parser.add_argument("--session-key", default="eval:direct:1")
    parser.add_argument("--no-guard", action="store_true", help="Disable Guard (transparent proxy).")
    parser.add_argument("--datamark", action="store_true", help="Enable datamarking of untrusted results.")
    args = parser.parse_args()

    result = asyncio.run(_run(
        args.config,
        args.prompt,
        args.session_key,
        guard_on=not args.no_guard,
        datamark_on=args.datamark,
    ))
    print(result.get("response") or "(no response)")


if __name__ == "__main__":
    main()

class _StubTool:
    """Minimal tool stub that satisfies the GuardedProxy interface."""

    def __init__(self, name: str, result: str = "ok") -> None:
        self.name = name
        self.schema: Any = None
        self.description: str | None = None
        self._result = result

    async def execute(self, **params: Any) -> Any:
        return self._result


class _StubRegistry:
    """Minimal ToolRegistry stub (no nanobot dependency)."""

    def __init__(self, tools: dict[str, Any] | None = None) -> None:
        self._tools: dict[str, Any] = tools or {}

    def register(self, tool: Any) -> None:
        self._tools[tool.name] = tool

    def unregister(self, name: str) -> None:
        self._tools.pop(name, None)

    def get(self, name: str) -> Any:
        return self._tools.get(name)

    @property
    def tool_names(self) -> list[str]:
        return list(self._tools.keys())


def _make_surface() -> SurfaceFile:
    """Minimal surface map — all tools absent so defaults apply."""
    return SurfaceFile(version=1, tools=[])

# ---------------------------------------------------------------------------
# Idempotence — double install must not double-wrap
# ---------------------------------------------------------------------------

def test_double_install_does_not_double_wrap() -> None:
    """Calling install_guard twice on the same registry leaves each tool wrapped once."""
    registry = _StubRegistry({
        "mcp_tool_x": _StubTool("mcp_tool_x"),
    })

    _install(registry)
    first_proxy = registry._tools["mcp_tool_x"]
    assert isinstance(first_proxy, GuardedProxy)

    # Second call — should be a no-op on the already-wrapped tool.
    _install(registry)
    second_proxy = registry._tools["mcp_tool_x"]

    # Same object: the proxy was not re-wrapped.
    assert second_proxy is first_proxy, (
        "install_guard called twice must not create a proxy-of-proxy"
    )
    # Still a single-layer GuardedProxy (not a GuardedProxy wrapping a GuardedProxy).
    assert isinstance(second_proxy, GuardedProxy)
    assert not isinstance(second_proxy._tool, GuardedProxy), (
        "inner _tool must be the original raw tool, not another GuardedProxy"
    )

def test_register_patch_blocks_forbidden() -> None:
    """After install_guard, registering 'spawn' or 'exec' is silently ignored."""

    assert "exec" not in registry._tools, "'exec' must not be re-registered after install"

    assert "mcp_tool_y" in registry._tools
