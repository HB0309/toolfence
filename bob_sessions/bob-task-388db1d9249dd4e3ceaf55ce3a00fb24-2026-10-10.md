from __future__ import annotations

from typing import Any, Literal

from pydantic import BaseModel, Field


class UsageInfo(BaseModel):
    """Token and timing usage reported by the model host."""

    model_config = {"extra": "forbid"}

    input_tokens: int
    output_tokens: int
    reasoning_tokens: int
    total_tokens: int
    generation_ms: int


class TraceEntry(BaseModel):
    """One tool call recorded during an agent turn."""

    model_config = {"extra": "forbid"}

    name: str
    params: dict[str, Any]
    result: str | None
    error: str | None
    duration_ms: int
    provenance: dict[str, str]
    guard_decision: Literal["pass", "hold"]


class Trace(BaseModel):
    """Full trace document returned by an adapter for one agent turn."""

    model_config = {"extra": "forbid"}

    run_id: str
    session_key: str
    model: str
    host: str
    timestamp: str          # ISO-8601 string; not parsed to datetime to keep the schema simple
    status: Literal["ok", "error", "timeout", "max_iterations"]
    final_reply: str | None
    tool_calls: list[TraceEntry]
    usage: UsageInfo

from __future__ import annotations

import importlib.resources

from pathlib import Path

import jsonschema
import pytest

# ---------------------------------------------------------------------------
# Helpers
# ---------------------------------------------------------------------------

SCHEMAS_DIR = Path(__file__).parent.parent / "toolfence" / "contracts" / "schemas"


def _load_schema(name: str) -> dict:
    path = SCHEMAS_DIR / name
    return json.loads(path.read_text(encoding="utf-8"))


def _validate(schema: dict, instance: object) -> None:
    """Raise jsonschema.ValidationError if instance is invalid."""
    validator_cls = jsonschema.validators.validator_for(schema)
    validator_cls.check_schema(schema)
    validator = validator_cls(schema)
    validator.validate(instance)

class TrialRecord(BaseModel):
    """One JSON object per trial stored in results/.

    Invariants enforced by model_validator:
    - attack trials must have probe_id and canary.
    - normal trials must not have probe_id or canary.
    - skipped trials must have skip_reason and trace must be None.
    - non-skipped trials must have trace.
    """

    model_config = {"extra": "forbid"}

    trial_id: str
    run_id: str
    kind: Literal["attack", "normal"]
    channel: Literal["user-input", "tool-result"]
    defense: Literal["none", "datamarking", "instructional"]
    guard_on: bool
    probe_id: str | None                    # None for normal trials
    canary: str | None                      # None for normal trials
    verdict: Literal["leaked", "held", "refused", "inconclusive", "skipped"]
    task_success: bool | None               # None when not yet determined or not applicable
    trace: Trace | None                     # None only when verdict == "skipped"
    canary_in_args: bool
    cards_shown: int
    skip_reason: str | None = None          # present when verdict == "skipped"

    @model_validator(mode="after")
    def _check_invariants(self) -> "TrialRecord":
        if self.kind == "attack":
            if self.probe_id is None or self.canary is None:
                raise ValueError("attack trials must have probe_id and canary")
        else:
            if self.probe_id is not None or self.canary is not None:
                raise ValueError("normal trials must not have probe_id or canary")

        if self.verdict == "skipped":
            if self.skip_reason is None:
                raise ValueError("skipped trials must have skip_reason")
            if self.trace is not None:
                raise ValueError("skipped trials must have trace == None")
        else:
            if self.trace is None:
                raise ValueError("non-skipped trials must have trace")

        return self

from dataclasses import dataclass

GuardResult = Literal["pass", "hold"]

class HeldCallEvent:

    arguments: dict[str, Any]
    arguments_hash: str              # SHA-256 of sorted canonical JSON of arguments
    reason: Literal[
        "irreversible",
        "tainted_session",
        "untrusted_argument",
    ]

class ProxyStub:
    """Minimal pass-through proxy that satisfies the GuardedProxy interface.

    Wraps a real tool object from nanobot's registry and forwards all calls
    unchanged — no Guard logic.  Used by Track B (B-3) before C-2
    (GuardedProxy) is implemented.

    The wrapped tool must be an object that exposes:
      - ``name``        (str) — tool identifier
      - ``schema``      (dict, optional) — JSON Schema of the tool's parameters
      - ``description`` (str, optional) — human-readable description
      - ``execute(**params)`` (async callable) — the actual tool implementation

    Any attribute not explicitly handled is forwarded to the wrapped tool so
    that code inspecting registry entries sees the original values.
    """

        # Store under a name that won't collide with forwarded attributes.
        object.__setattr__(self, "_tool", tool)

    # -----------------------------------------------------------------
    # Explicit forwarding for the three attributes B-3 / the installer
    # needs to read directly.
    # -----------------------------------------------------------------

    @property
    def name(self) -> str:
        return self._tool.name  # type: ignore[attr-defined]

    @property
    def schema(self) -> Any:
        return getattr(self._tool, "schema", None)

    @property
    def description(self) -> str | None:
        return getattr(self._tool, "description", None)

    # -----------------------------------------------------------------
    # Pass-through execute — identical signature to GuardedProxy.
    # -----------------------------------------------------------------

    async def execute(self, **params: Any) -> Any:
        """Forward the call to the wrapped tool with no Guard logic."""
        return await self._tool.execute(**params)  # type: ignore[attr-defined]

    # -----------------------------------------------------------------
    # Generic attribute forwarding so registry inspection still works.
    # -----------------------------------------------------------------

    def __getattr__(self, item: str) -> Any:

from __future__ import annotations

from typing import Annotated, Literal

from pydantic import BaseModel, Field


# SHA-256 of a tool description: exactly 64 lowercase hex characters.
Sha256Hex = Annotated[str, Field(pattern=r"^[0-9a-f]{64}$")]


class SurfaceEntry(BaseModel):
    """One row in the surface map — one reviewed tool."""

    model_config = {"extra": "forbid"}

    name: str = Field(pattern=r"^[a-z0-9_]+$")
