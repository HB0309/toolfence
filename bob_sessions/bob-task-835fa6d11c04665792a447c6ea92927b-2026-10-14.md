### Problem 2 — `scan_openapi` builds wrong names — they never match nanobot

**What's wrong:** [`SurfaceScanner.scan_openapi`](toolfence/engine/surface.py:207) builds tool names from the OpenAPI path+method:

```python
clean_path = path.replace("{", "").replace("}", "")
segments = [s for s in clean_path.split("/") if s]
name_parts = [method.lower()] + segments
tool_name = "_".join(name_parts).replace("-", "_")
```

For `POST /cancel/{booking_id}` this yields `post_cancel_booking_id`.  
Nanobot actually names the tool `mcp_galaxium_cancel_booking_endpoint_cancel_booking_id_post`.

The name nanobot gives to a Galaxium tool comes from FastAPI's **function-name-based** `operationId` (`cancel_booking_endpoint_cancel__booking_id__post` → sanitised → `cancel_booking_endpoint_cancel_booking_id_post`), then prefixed with `mcp_galaxium_`. `scan_openapi` has no knowledge of the `operationId` field; it re-derives the name from the HTTP path, producing a completely different string.

Additionally, the `_is_valid_name` guard at [`surface.py:347`](toolfence/engine/surface.py:347) rejects names containing `-`. `scan_openapi` does call `.replace("-", "_")` but that's applied *after* the `_join`, so `internal/bookings/from-hold` → `post_internal_bookings_from_hold` — that one would pass. But crucially, the names it produces still don't carry the `mcp_galaxium_` prefix or the FastAPI `operationId` fragment, so they will **never** match nanobot's live registry.

The correct approach for OpenAPI-based scanning of Galaxium would be to read the `operationId` field directly from each path/method entry, which `scan_openapi` currently ignores entirely.

**File & line:** [`toolfence/engine/surface.py:228–252`](toolfence/engine/surface.py:228) (`scan_openapi` body)  
**Track:** **A** (surface scanner is Track A's code) — but the mismatch blocks B-5 output, so B needs to flag it to A or work around it by using `scan_json` with the live tool list from nanobot instead of relying on `scan_openapi`.

async def _dump(config_path: str) -> list[tuple[str, str, str]]:
    """Return list of (name, description, sha256) for every tool in the registry."""
    from nanobot.agent.loop import AgentLoop
    from nanobot.agent.tools.mcp import MCPProvider
    from nanobot.agent.tools.registry import ToolRegistry
    from nanobot.bus.queue import MessageBus
    from nanobot.cli.runtime_config import _load_runtime_config
    from nanobot.providers.factory import make_provider

    workspace = tempfile.mkdtemp(prefix="toolfence-dump-")
    try:
        runtime_config = _load_runtime_config(config_path, workspace)

        bus = MessageBus()
        tools = ToolRegistry()
        mcp_provider = MCPProvider.from_config(runtime_config, tools)
        provider = make_provider(runtime_config)

        agent_loop = AgentLoop.from_config(
            runtime_config,
            bus,
            provider=provider,
            tool_registry=tools,
        )

        try:
            await mcp_provider.connect()
        except Exception as exc:
            print(
                f"ERROR: Could not connect to MCP server — is Galaxium running?\n  {exc}",
                file=sys.stderr,
            )
            await agent_loop.aclose()
            sys.exit(1)

        try:
            # Install Guard (removes spawn, wraps everything else).
            # Pass an empty surface — we only need the side-effect of spawn removal.
            empty_surface = SurfaceFile(version=1, tools=[])
            install_guard(
                tools,
                empty_surface,
                ProvenanceStore(),
                SessionTaint(),
                GuardDecision(),
                _noop_on_held,
                ConfirmGrant(),
                guard_on=False,   # transparent — we only want spawn removed
                datamark_on=False,
            )

            results: list[tuple[str, str, str]] = []
            for name in sorted(tools.tool_names):
                tool = tools.get(name)
                desc = tool.description if tool is not None else ""
                sha = description_sha256(desc)
                results.append((name, desc, sha))
            return results
        finally:
            await agent_loop.aclose()
            await mcp_provider.aclose()
    finally:
        shutil.rmtree(workspace, ignore_errors=True)


def main() -> None:
    parser = argparse.ArgumentParser(
        description="Dump tool names + descriptions from a live nanobot+Galaxium session."
    )
    parser.add_argument(
        "--config", default=_DEFAULT_CONFIG,
        help=f"nanobot config JSON (default: {_DEFAULT_CONFIG})",
    )
    parser.add_argument(
        "--output", default=None,
        help="Write output to file instead of stdout",
    )
    args = parser.parse_args()

    entries = asyncio.run(_dump(args.config))

    lines: list[str] = []
    mcp_entries = [(n, d, s) for n, d, s in entries if n.startswith("mcp_")]
    builtin_entries = [(n, d, s) for n, d, s in entries if not n.startswith("mcp_")]

    lines.append(f"# {len(entries)} tools total  "
                 f"({len(mcp_entries)} MCP, {len(builtin_entries)} built-in)\n")

    lines.append("## MCP tools\n")
    for name, desc, sha in mcp_entries:
        lines.append(f"name:   {name}")
        lines.append(f"sha256: {sha}")
        lines.append(f"desc:\n{desc}")
        lines.append("---")

    lines.append("\n## Built-in tools\n")
    for name, desc, sha in builtin_entries:
        lines.append(f"name:   {name}")
        lines.append(f"sha256: {sha}")
        lines.append(f"desc:\n{desc}")
        lines.append("---")

    output = "\n".join(lines) + "\n"

    if args.output:
        Path(args.output).write_text(output, encoding="utf-8")
        print(f"Written to {args.output}")
    else:
        print(output)


if __name__ == "__main__":
    main()

def _parse_dump(path: Path) -> dict[str, str]:
    """Parse tool_dump.txt and return {name: sha256} for every entry."""
    name_re = re.compile(r"^name:\s+(\S+)$")
    sha_re = re.compile(r"^sha256:\s+([0-9a-f]{64})$")
    result: dict[str, str] = {}
    current_name: str | None = None
    for line in path.read_text(encoding="utf-8").splitlines():
        m = name_re.match(line)
        if m:
            current_name = m.group(1)
            continue
        m = sha_re.match(line)
        if m and current_name is not None:
            result[current_name] = m.group(1)
            current_name = None
    return result
